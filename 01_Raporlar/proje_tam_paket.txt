  NIHAI SECIM: EGARCH(1,1) x Skewed-t

  SECIM GERCEKLESI:
  1. AIC: EGARCH x skewt en iyi (175,063.06, GJR'den 219 puan iyi)
  2. BIC: EGARCH x t en iyi (175,129.88, skewt 3 puan geride)
  3. Kalicilik: EGARCH durgun (0.948), GJR sinirda (1.000)
  4. Katsiampa (2017): EGARCH Bitcoin icin en iyi — tutarli
  5. Asimetri: Gamma anlami (t/skewt'te, p<0.01)
  6. OOS: DM testi anlamsiz (p=0.595), secim robust
  7. Mean equation tutarsizligi: ORTADAN KALKTI (AR(2) her yerde)

  Model Denklemi (Nelson 1991):
  log(h_t) = omega + alpha*(|z_{t-1}| - E|z|) + gamma*z_{t-1} + beta*log(h_{t-1})
  z_t = Const + phi1*z_{t-1} + phi2*z_{t-2} + e_t    [AR(2)]
  e_t ~ Skewed-t(df=eta, skew=lambda)

  Katsayilar (tam veri):
  - omega = 0.063 (log-varyans sabiti)
  - alpha = 0.365 (boyut etkisi)
  - beta  = 0.948 (kalıcılık)
  - gamma = -0.027 (leverage etkisi, t'de anlami)
  - eta   = 3.29 (serbestlik)
  - lambda = -0.026 (sola kayma)

  OVERFITTING TARTISMASI:
  EGARCH in-sample'de en iyi (AIC=175,063) ama OOS'de QLIKE'da
  sonuncu (1.843). Bu durum olasi asiri uyumu (overfitting)
  gosterebilir:

  1. EGARCH'in 9 parametresi var (digerlerinde 7-8). Fazla
     parametre, sample-fit'i iyilestirir ama generalizasyonu
     zayiflatabilir.
  2. QLIKE metrigi dogrudan volatilite tahmininin kalitesini
     olcer (log-konik skor). EGARCH'in RMSE'si en iyi ama
     QLIKE'si en kotu — bu, nokta tahmininin iyi olup
     dagilim tahmininin zayif kalabilecegini gosterebilir.
  3. DM testi anlamsiz (p=0.595): farklar istatistiksel olarak
     anlamsiz, yani overfitting iddiasi kesin degil.
  4. Sonuc: EGARCH in-sample'de tercih edilir ama OOS'de
     farklar anlamsizdir. Pratikte fark yoktur.

  NOT: EGARCH OOS'de en iyi RMSE'ye de sahip (5.880). Yani
  nokta tahmini olarak en iyi, risk olcumu olarak en kotu.

def kur_model(r, vol_tipi, dist_tipi, igarch=False):
    """Tek bir GARCH ailesi modeli kurar ve sonuclari dondurur."""
    vol_kutle = {"ARCH": {"p": 10, "q": 0},
                 "GARCH": {"p": 1, "q": 1},
                 "EGARCH": {"p": 1, "q": 1, "o": 1},
                 "GJR": {"p": 1, "q": 1, "o": 1}}
    dk = vol_kutle.get(vol_tipi)
    if dk is None:
        return None

    if igarch:
        # Approx IGARCH: GARCH(1,1) kur, alpha+beta=1'e yakin mi kontrol et
        # NOT: Bu GERCEK sinirli MLE degil (alpha+beta=1 kismi dayatilmadi)
        # Sadece standart GARCH ciktisini "yaklasik IGARCH" olarak etiketler
        sonuc = {"vol": "Approx_IGARCH", "dist": dist_tipi, "model_tipi": "Approx_IGARCH",
                 "yakinsadi": False, "aic": np.nan, "bic": np.nan,
                 "loglik": np.nan, "k": np.nan, "omega": 0.0}
        try:
            m = arch_model(r * 100, mean="AR", lags=2,
                           vol="Garch", p=1, q=1, o=0,
                           dist=dist_tipi)
            f = m.fit(disp="off", show_warning=False)
            if f.convergence_flag != 0:
                return sonuc
            # alpha+beta=1'e yakin mi kontrol et
            a = f.params.get("alpha[1]", 0)
            b = f.params.get("beta[1]", 1)
            yaklasik = abs(a + b - 1.0)
            if yaklasik < 0.05:  # %5 tolerans
                sonuc["yakinsadi"] = True
                sonuc["aic"] = f.aic
                sonuc["bic"] = f.bic
                sonuc["loglik"] = f.loglikelihood
                sonuc["k"] = f.params.shape[0]
                sonuc["omega"] = 0.0
                sonuc["alpha"] = a
                sonuc["beta"] = b
                sonuc["kalicilik"] = a + b
                sonuc["gamma"] = 0.0
                if dist_tipi == "t":
                    sonuc["nu"] = f.params.get("nu", np.nan)
                elif dist_tipi == "skewt":
                    sonuc["nu"] = f.params.get("nu", np.nan)
                    sonuc["lambda"] = f.params.get("lambda", np.nan)
            return sonuc
        except:
            return sonuc

    sonuc = {"vol": vol_tipi, "dist": dist_tipi, "model_tipi": "GARCH_AILESI",
             "yakinsadi": False, "aic": np.nan, "bic": np.nan,
             "loglik": np.nan, "k": np.nan, "omega": np.nan}
    try:
        # arch kutuphanesinde vol: "Garch" (GARCH, GJR, ARCH icin) veya "EGARCH"
        vol_arch = "EGARCH" if vol_tipi == "EGARCH" else "Garch"
        m = arch_model(r * 100, mean="AR", lags=2,
                       vol=vol_arch, **dk, dist=dist_tipi)
        f = m.fit(disp="off", show_warning=False)
        if f.convergence_flag != 0:
            return sonuc
        sonuc["yakinsadi"] = True
        sonuc["aic"] = f.aic
        sonuc["bic"] = f.bic
        sonuc["loglik"] = f.loglikelihood
        sonuc["k"] = f.params.shape[0]
        for p_adi in ["omega", "alpha[1]", "beta[1]", "gamma[1]"]:
            sonuc[p_adi.split("[")[0]] = f.params.get(p_adi, np.nan)

        a = sonuc.get("alpha", sonuc.get("alpha[1]", 0))
        b = sonuc.get("beta", sonuc.get("beta[1]", 0))
        g = sonuc.get("gamma", 0)
        if np.isnan(g):
            g = 0
        if vol_tipi == "GJR":
            sonuc["kalicilik"] = a + b + 0.5 * g
            sonuc["kalicilik_ab"] = a + b
        elif vol_tipi == "EGARCH":
            # EGARCH'da kalicilik sadece beta'dir (log-varyans modeli)
            sonuc["kalicilik"] = b
            sonuc["kalicilik_ab"] = b
        else:
            sonuc["kalicilik"] = a + b
            sonuc["kalicilik_ab"] = a + b
        if dist_tipi == "t":
            sonuc["nu"] = f.params.get("nu", np.nan)
        elif dist_tipi == "skewt":
            sonuc["nu"] = f.params.get("nu", np.nan)
            sonuc["lambda"] = f.params.get("lambda", np.nan)
        return sonuc
    except Exception as e:
        sonuc["hata"] = str(e)[:80]
        return sonuc

        e2 = lm_test ** 2
        # Basit ARCH-LM: Regresyon e2 ~ sabit + e2_lag1 + ... + e2_lag_m
        from statsmodels.regression.linear_model import OLS
        from statsmodels.tools import add_constant
        y_dep = e2.values[m_val:]
        X_mat = np.column_stack([e2.values[m_val - i:-i] if i > 0 else e2.values[m_val:]
                                 for i in range(1, m_val + 1)])
        X_mat = add_constant(X_mat)
        model_ols = OLS(y_dep, X_mat).fit()
        # TR² test
        n_obs = len(y_dep)
        lm_stat = n_obs * model_ols.rsquared
        from scipy.stats import chi2
        p_val = 1 - chi2.cdf(lm_stat, m_val)
        yaz("  m=%2d:  LM=%.2f  p=%.2e  %s" % (m_val, lm_stat, p_val,

def oku():
    df = pd.read_csv("bitcoin_saatlik_temiz_v4.csv", index_col=0, parse_dates=True)
    return df["Log_Return"].dropna()

def model_adi(vol, dist):
    return "%s_%s" % (vol, dist)

def fit_garch(r, vol, dist, lags=2):
    """GJR-GARCH, EGARCH, GARCH, veya IGARCH kurar."""
    o = 1 if vol in ("GJR", "EGARCH") else 0
    v = "Garch" if vol in ["GARCH", "GJR"] else vol
    if vol == "IGARCH":
        v = "Garch"; o = 0
    try:
        m = arch_model(r * 100, mean="AR", lags=lags, vol=v, p=1, q=1, o=o, dist=dist)
        f = m.fit(disp="off", show_warning=False)
        return f if f.convergence_flag == 0 else None
    except:
        return None

def kalicilik_hesapla(f, vol):
    p = f.params
    a = p.get("alpha[1]", 0); b = p.get("beta[1]", 0); g = p.get("gamma[1]", 0)
    if vol == "GJR":
        return a + b + 0.5 * g
    elif vol == "EGARCH":
        # EGARCH'da kalicilik sadece beta'dir (log-varyans modeli)
        return b
    return a + b

    adi = model_adi(vol, dist)
    yaz("\n  --- %s Rolling OOS (pencere=%d, refit her %d) ---" % (adi, PENCERE, REFIT_HER))
    t_start = time.time()

    tahminler = pd.Series(index=test.index, dtype=float)
    son_fit = None
    son_params = None

    fit_sayisi = 0
    for i in range(len(test)):
        # Train verisini belirle
        test_bas = train_sinir + i
        train_son = test_bas
        train_bas = max(0, test_bas - PENCERE)
        r_pencere = r.iloc[train_bas:train_son]

        # Refit gerekli mi?
        if i % REFIT_HER == 0 or son_fit is None:
            son_fit = fit_garch(r_pencere, vol, dist)
            fit_sayisi += 1
            if son_fit is None:
                tahminler.iloc[i] = np.nan
                continue

        # One-step-ahead tahmin
        if son_fit is not None:
            try:
                tahmin_var = son_fit.forecast(horizon=1, reindex=False).variance.iloc[-1].values[0]
                tahminler.iloc[i] = tahmin_var / 10000  # yuzde kare -> orijinal olcek
            except:
                tahminler.iloc[i] = np.nan

    tahminler = tahminler.dropna()
    ort_fit = fit_sayisi
    gecen = time.time() - t_start
    yaz("  Tamamlandi: %d tahmin, %d fit, %.1f saniye" % (len(tahminler), ort_fit, gecen))

    # Metrikler hesapla
    ortak_idx = realized_var.index.intersection(tahminler.index)
    if len(ortak_idx) < 10:
        yaz("  [!] Yeterli ortak gozlem yok (%d), atlandi" % len(ortak_idx))
        continue

    rv = realized_var.loc[ortak_idx].values
    tv = tahminler.loc[ortak_idx].values
    tv = np.maximum(tv, 1e-10)  # sifir bolme korumasi

    rmse = np.sqrt(np.mean((rv - tv) ** 2))
    mae = np.mean(np.abs(rv - tv))
    qlike = np.mean(np.log(rv / tv) + tv / rv - 1)

    oos_sonuclar.append({
        "Model": adi, "RMSE": rmse, "MAE": mae, "QLIKE": qlike,
        "tahmin_sayisi": len(tahminler), "fit_sayisi": ort_fit
    })
    tahmin_seri[adi] = tahminler.loc[ortak_idx]

    yaz("  RMSE=%.6f  MAE=%.6f  QLIKE=%.6f" % (rmse, mae, qlike))

# Diebold-Mariano Testi
yaz("\n  DIEBOLD-MARIANO TESTI (Newey-West HAC duzeltmeli, ilk 2 model arasi):")
if len(oos_df) >= 2:
    m1_adi = oos_df.iloc[0]["Model"]
    m2_adi = oos_df.iloc[1]["Model"]
    if m1_adi in tahmin_seri and m2_adi in tahmin_seri:
        ortak = tahmin_seri[m1_adi].index.intersection(tahmin_seri[m2_adi].index)
        ortak = ortak.intersection(realized_var.index)
        if len(ortak) > 10:
            rv_dm = realized_var.loc[ortak].values
            t1 = tahmin_seri[m1_adi].loc[ortak].values
            t2 = tahmin_seri[m2_adi].loc[ortak].values
            t1 = np.maximum(t1, 1e-10)
            t2 = np.maximum(t2, 1e-10)

            # QLIKE kayip fonksiyonu
            L1 = np.log(rv_dm / t1) + t1 / rv_dm - 1
            L2 = np.log(rv_dm / t2) + t2 / rv_dm - 1
            d = L1 - L2
            d_bar = np.mean(d)
            # HAC standart hata (Newey-West, otomatik lag secimi)
            T = len(d)
            # Newey-West: otomatik lag sayisi (Andrews 1991: floor(4*(T/100)^(2/9)))
            max_lag = max(1, int(np.floor(4 * (T / 100) ** (2/9))))
            gamma0 = np.var(d, ddof=1)
            hac_var = gamma0 / T
            # Otokorelasyonlu lag'lari ekle
            for lag in range(1, max_lag + 1):
                weight = 1 - lag / (max_lag + 1)  # Bartlett agirlik
                gamma_lag = np.mean((d[:-lag] - d_bar) * (d[lag:] - d_bar))
                hac_var += 2 * weight * gamma_lag / T
            dm_stat = d_bar / np.sqrt(max(hac_var, 1e-20))
            dm_p = 2 * (1 - stats.norm.cdf(abs(dm_stat)))
            yaz("  %s vs %s" % (m1_adi, m2_adi))
            kazanan = m1_adi if d_bar < 0 else m2_adi
            yaz("  DM istatistigi: %.4f  p-degeri: %.4f" % (dm_stat, dm_p))
            yaz("  Kazanan: %s (%s)" % (kazanan,
                "anlamlı" if dm_p < 0.05 else "anlamsiz (p>0.05)"))
            yaz("  Not: QLIKE'ye gore kucuk = daha iyi")
        else:
            yaz("  Yeterli ortak gozlem yok")
    else:
        yaz("  Tahmin serileri bulunamadi")

    if np.isnan(lam):

    # Parametrik VaR (Student-t varsayimiyla)
    if nu is None:
        yaz("\n  VaR hesaplanamiyor (nu parametresi bulunamadi).")
    else:
        # VaR_alpha = mu + sigma * t_quantile(alpha, df=nu)
        mu = 0  # constant mean ~ 0
        t_q95 = stats.t.ppf(0.05, df=nu)  # %95 VaR icin sol quantile
        t_q99 = stats.t.ppf(0.01, df=nu)  # %99 VaR icin

        # Rolling VaR hesaplama
        PENCERE_VaR = 1000
        test_bas_idx = int(len(r) * 0.80)
        test_veri = r.iloc[test_bas_idx:]
        test_sigma = sigma_koşullu.iloc[test_bas_idx:]

        VaR_95 = pd.Series(index=test_veri.index, dtype=float)
        VaR_99 = pd.Series(index=test_veri.index, dtype=float)

        fit_sayisi_var = 0
        for i in range(len(test_veri)):
            test_bas = test_bas_idx + i
            train_bas = max(0, test_bas - PENCERE_VaR)

                    tahmin_sigma = np.sqrt(tahmin_var) / 100
                    VaR_95.iloc[i] = mu + tahmin_sigma * t_q95
                    VaR_99.iloc[i] = mu + tahmin_sigma * t_q99

                    VaR_99.iloc[i] = np.nan

            gercek = test_veri.loc[ortak].values

            ihlal = gercek < tahmin  # VaR'in altina dustu mu?
            n_ihlal = np.sum(ihlal)
            n_toplam = len(ihlal)
            orani = n_ihlal / n_toplam
            beklenen_oran = 1 - seviye / 100.0

            # Kupiec LR testi
            if n_ihlal > 0 and n_ihlal < n_toplam:
                p_hat = n_ihlal / n_toplam
                p0 = beklenen_oran
                LR = -2 * (n_toplam * np.log(1 - p0) + n_ihlal * np.log(p0)
                           - n_toplam * np.log(1 - p_hat) - n_ihlal * np.log(p_hat))
                kupiec_p = 1 - chi2.cdf(LR, 1)
            else:
                LR = 0; kupiec_p = 1.0

            durum = "KABUL" if kupiec_p > 0.05 else "RED"
            yaz("  %d%% VaR:  ihlal=%d/%d (%.2f%%, beklenen=%.1f%%)  LR=%.2f  p=%.4f  %s" % (
                seviye, n_ihlal, n_toplam, orani * 100, beklenen_oran * 100, LR, kupiec_p, durum))

        # VaR grafigi
